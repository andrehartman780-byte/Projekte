Function Specifications

Struct: Book

Define a struct with owned data. Use these exact field names and types: 

pub struct Book {
    pub id: u32,
    pub title: String,
    pub author: String,
    pub year: u16,
}

Keep the struct as-is. The tests access the fields directly.

Book::new

pub fn new(id: u32, title: &str, author: &str, year: u16) -> Self
  • - Use title and author string slices to build owned Strings.
  • - Return a fully initialized Book.

Book::title

pub fn title(&self) -> &str
  • - Return a string slice that points into self.title.
  • - Do not allocate a new String.

build_index

pub fn build_index(books: Vec<Book>) -> HashMap<u32, Book>
  • - Create a HashMap keyed by id.
  • - Move each Book into the map (do not clone).

get_book

pub fn get_book(index: &HashMap<u32, Book>, id: u32) -> Option<&Book>
  • - Return Some(&Book) if the id exists.
  • - Return None if it does not.

parse_id

pub fn parse_id(input: &str) -> Result<u32, String>
  • - Parse a u32 from input.
  • - On failure, return Err with a clear message.
  • - Do not panic.

filter_titles

pub fn filter_titles<'a, F>(
    books: &'a [Book],

where
    F: Fn(&Book) -> bool,

  • - Return a Vec of &str titles (string slices).
  • - Do not allocate new Strings for the titles.

title_parts

pub fn title_parts(title: &str) -> (&str, Option<&str>)
  • - Split on the first :.
  • - Return the part before the colon.
